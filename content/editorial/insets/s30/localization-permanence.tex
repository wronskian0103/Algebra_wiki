Here $A$ is an integral domain, $1\in S$, $S$ is multiplicatively closed,
and $0\notin S$. Thus $S^{-1}A$ embeds in the fraction field and is a
nonzero domain.

\begin{proposition}\label{pid-localization}
If $A$ is a PID, then $S^{-1}A$ is a PID.
\end{proposition}
\begin{proof}\label{pid-localization-proof}
Every localized ideal is the extension of its contraction $I$.
Writing $I=(a)$ shows that its extension is generated by $a/1$.
Thus every ideal of the localized domain is principal.
\end{proof}
\begin{lemma}\label{prime-localization}
For a prime element $p$ of $A$, its image $p/1$ is either a unit or
a prime element. It is a unit exactly when $(p)\cap S\ne\varnothing$.
\end{lemma}
\begin{proof}\label{prime-localization-proof}
If $s=pa\in S$, then $a/s$ is an inverse of $p/1$.
Conversely, $(p/1)(a/s)=1$ implies $pa=s$ in the fraction field.
When $(p)\cap S$ is empty, its extension is a proper prime ideal by
\wikilink{s28--b005}{prime-ideal correspondence}, and is generated by $p/1$.
\end{proof}
\begin{theorem}\label{ufd-localization}
If $A$ is a UFD, then $S^{-1}A$ is a UFD.
\end{theorem}
\begin{proof}\label{ufd-localization-proof}
Factor the numerator of a nonzero fraction into a unit and primes in $A$.
The denominator and any prime factors whose ideals meet $S$ become units.
The remaining factors are prime by the lemma. Thus every nonzero
nonunit is a product of primes. To prove uniqueness, a prime in one
factorization divides a factor in the other, hence is associate to it;
cancel in the domain and continue. This is
\wikilink{s29--b007}{unique factorization}.
\end{proof}
\begin{example}\label{integer-localizations}
For a prime integer $p$, both $\mathbb Z[1/p]$ and $\mathbb Z_{(p)}$
are PIDs. In the former $p$ becomes a unit; in the latter the integers
not divisible by $p$ become units. A localization may be a field,
which is still a PID and a UFD.
\end{example}
